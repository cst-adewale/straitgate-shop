<script>
 const auth = document.getElementById('auth');
 auth.open();
 auth.addEventListener('auth:submit', async e => {
   const res = await fetch('/api/login', {method:'POST', body: JSON.stringify(e.detail)});
   res.ok ? auth.close() : auth.setError('Incorrect email or password.');
 });
</script>
